# Common TypeScript Types Reference

Reusable type patterns for JavaScript to TypeScript migration. These types are auto-generated by `generate-common-types.js`.

## API Response Types

### Basic API Response

```typescript
interface ApiResponse<T = unknown> {
  data: T;
  status: number;
  message: string;
  success: boolean;
}

// Usage
interface User {
  id: number;
  name: string;
}

const response: ApiResponse<User> = await fetch('/api/users/1').then(r => r.json());
console.log(response.data.name);
```

### Paginated API Response

```typescript
interface ApiListResponse<T> extends ApiResponse<T[]> {
  total: number;
  page: number;
  pageSize: number;
}

// Usage
const usersResponse: ApiListResponse<User> = await fetchUsers({ page: 1, pageSize: 10 });
console.log(`Showing ${usersResponse.data.length} of ${usersResponse.total} users`);
```

### Error Response

```typescript
interface ApiError {
  code: string;
  message: string;
  details?: Record<string, unknown>;
  statusCode: number;
}

// Usage in catch block
try {
  await apiCall();
} catch (error) {
  const apiError = error as ApiError;
  console.error(`${apiError.code}: ${apiError.message}`);
}
```

### Async Result Pattern

```typescript
interface AsyncResult<T, E = Error> {
  data?: T;
  error?: E;
  isLoading: boolean;
}

// Usage with React state
const [state, setState] = useState<AsyncResult<User>>({ isLoading: true });

// Usage in API calls
async function fetchUser(id: number): Promise<AsyncResult<User>> {
  try {
    const data = await api.get(`/users/${id}`);
    return { data, isLoading: false };
  } catch (error) {
    return { error: error as Error, isLoading: false };
  }
}
```

### Result Type (Rust-style)

```typescript
type Result<T, E = Error> = 
  | { success: true; data: T }
  | { success: false; error: E };

// Usage
function divide(a: number, b: number): Result<number, string> {
  if (b === 0) {
    return { success: false, error: 'Division by zero' };
  }
  return { success: true, data: a / b };
}

const result = divide(10, 2);
if (result.success) {
  console.log(result.data); // 5
} else {
  console.error(result.error);
}
```

## Pagination Types

### Offset-based Pagination

```typescript
interface PaginationParams {
  page: number;
  pageSize: number;
  offset?: number;
}

interface PaginationMeta {
  currentPage: number;
  pageSize: number;
  totalItems: number;
  totalPages: number;
  hasNextPage: boolean;
  hasPreviousPage: boolean;
}

interface PaginatedResponse<T> {
  items: T[];
  pagination: PaginationMeta;
}

// Usage
async function getProducts(params: PaginationParams): Promise<PaginatedResponse<Product>> {
  const offset = (params.page - 1) * params.pageSize;
  // ... fetch from API
}
```

### Cursor-based Pagination

```typescript
interface CursorPaginationParams {
  cursor?: string;
  limit: number;
}

interface CursorPaginatedResponse<T> {
  items: T[];
  nextCursor?: string;
  previousCursor?: string;
  hasMore: boolean;
}

// Usage (infinite scroll)
async function loadMore(cursor?: string): Promise<CursorPaginatedResponse<Post>> {
  return api.get('/posts', { cursor, limit: 20 });
}
```

### Custom Error Classes

```typescript
class BaseError extends Error {
  constructor(
    message: string,
    public code: string,
    public statusCode: number = 500,
    public context?: Record<string, unknown>
  ) {
    super(message);
    this.name = 'BaseError';
  }
}

class NotFoundError extends BaseError {
  constructor(resource: string) {
    super(`${resource} not found`, 'NOT_FOUND', 404);
    this.name = 'NotFoundError';
  }
}

class UnauthorizedError extends BaseError {
  constructor(message = 'Unauthorized') {
    super(message, 'UNAUTHORIZED', 401);
    this.name = 'UnauthorizedError';
  }
}

// Usage
if (!user) {
  throw new NotFoundError('User');
}
```

### Base Event

```typescript
interface BaseEvent {
  type: string;
  timestamp: string | Date;
  source?: string;

}
```

### Domain Event (CQRS/Event Sourcing)

```typescript
interface DomainEvent<T = unknown> extends BaseEvent {
  aggregateId: string;
  aggregateType: string;
  payload: T;
  version: number;
}

// Usage
interface UserCreatedPayload {
  userId: string;
  email: string;
  name: string;
}

const event: DomainEvent<UserCreatedPayload> = {
  type: 'USER_CREATED',
  timestamp: new Date(),
  aggregateId: 'user-123',
  aggregateType: 'User',
  payload: { userId: 'user-123', email: 'test@example.com', name: 'Test' },
  version: 1
};
```

### Event Handler

```typescript
type EventHandler<T extends BaseEvent> = (event: T) => Promise<void> | void;

// Usage
const handlers: Map<string, EventHandler<BaseEvent>[]> = new Map();

function subscribe<T extends BaseEvent>(eventType: string, handler: EventHandler<T>) {
  const existing = handlers.get(eventType) || [];
  handlers.set(eventType, [...existing, handler as EventHandler<BaseEvent>]);
}
```

## React-Specific Types

### Component Props Utilities

```typescript
import type { ReactNode, CSSProperties } from 'react';

type WithChildren<T = object> = T & { children?: ReactNode };
type WithClassName<T = object> = T & { className?: string };
type WithStyle<T = object> = T & { style?: CSSProperties };

// Combined
type BaseComponentProps = WithChildren<WithClassName<WithStyle>>;

// Usage
function Card({ children, className, style }: BaseComponentProps) {
  return <div className={className} style={style}>{children}</div>;
}
```

### Async State

```typescript
interface AsyncState<T, E = Error> {
  data: T | null;
  error: E | null;
  isLoading: boolean;
  isError: boolean;
  isSuccess: boolean;
}

// Hook usage
function useAsync<T>(asyncFn: () => Promise<T>): AsyncState<T> {
  const [state, setState] = useState<AsyncState<T>>({
    data: null,
    error: null,
    isLoading: true,
    isError: false,
    isSuccess: false
  });
  
  useEffect(() => {
    asyncFn()
      .then(data => setState({ data, isLoading: false, isError: false, isSuccess: true, error: null }))
      .catch(error => setState({ data: null, isLoading: false, isError: true, isSuccess: false, error }));
  }, []);
  
  return state;
}
```

### Event Handlers

```typescript
type ChangeHandler<T = HTMLInputElement> = (event: React.ChangeEvent<T>) => void;
type SubmitHandler = (event: React.FormEvent<HTMLFormElement>) => void;
type ClickHandler<T = HTMLElement> = (event: React.MouseEvent<T>) => void;

// Usage
function MyForm() {
  const handleChange: ChangeHandler = (e) => {
    console.log(e.target.value);
  };
  
  const handleSubmit: SubmitHandler = (e) => {
    e.preventDefault();
    // submit
  };
  
  return (
    <form onSubmit={handleSubmit}>
      <input onChange={handleChange} />
    </form>
  );
}
```

### Express Middleware

```typescript
import type { Request, Response, NextFunction } from 'express';

type AsyncRequestHandler = (
  req: Request,
  res: Response,
  next: NextFunction
) => Promise<void | Response>;

type Middleware = RequestHandler | AsyncRequestHandler;

// Usage
const authMiddleware: AsyncRequestHandler = async (req, res, next) => {
  const token = req.headers.authorization;
  if (!token) {
    return res.status(401).json({ error: 'Unauthorized' });
  }
  req.user = await verifyToken(token);
  next();
};
```

## Utility Types

### Nullable

```typescript
type Nullable<T> = T | null;

// Usage
let userName: Nullable<string> = null;
userName = 'John';
```

### Deep Partial

```typescript
type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends object ? DeepPartial<T[P]> : T[P];
};

// Usage
interface Config {
  server: {
    host: string;
    port: number;
  };
}

const partialConfig: DeepPartial<Config> = {
  server: { port: 3000 } // host is optional
};
```

### Deep Required

```typescript
type DeepRequired<T> = {
  [P in keyof T]-?: T[P] extends object ? DeepRequired<T[P]> : T[P];
};
```

### Make Optional

```typescript
type Optional<T, K extends keyof T> = Omit<T, K> & Partial<Pick<T, K>>;

// Usage
interface User {
  id: number;
  name: string;
  email: string;
}

type UserUpdate = Optional<User, 'id'>;

const update: UserUpdate = { name: 'New Name' }; // id is optional
```

### Make Required

```typescript
type Required<T, K extends keyof T> = Omit<T, K> & Required<Pick<T, K>>;

// Usage
type UserWithRequiredEmail = Required<User, 'email'>;
```
